#include<iostream>
 
using namespace std;
 
template<class T>
class Queue {
public:
    T *buff;
 
    explicit Queue(int m1) {
        Max = m1;
        now = 0;
        buff = new T[Max];
        cout << "Queue constructor called!" << endl;
    }
 
    Queue(const Queue &q1) {
        Max = q1.Max;
        now = q1.now;
        buff = new T[Max];
        cout << "Queue copy constructor called!" << endl;
    }
 
    ~Queue() {
        delete[] buff;
        cout << "Queue destructor called!" << endl;
    }
 
    void In(const T &t1) {
        if (now >= Max) {
            cout << "the queue is full!" << endl;
        } else {
            buff[++now] = t1;
            // Tail = &buff[now];
        }
    }
 
    void Out(int n) {
        if (now == 0)
            cout << "the queue is empty,can't out!" << endl;
        else {
            T temp;
            int i;
            for (i = n; i < Max; i++) {
                buff[i - 1] = buff[i];
            }
        }
    }
 
    bool Empty() {
        if (now == 0) {
            cout << "the queue is empty!" << endl;
            return true;
        } else {
            cout << "the queue isn't empty!" << endl;
            return false;
        }
    }
 
private:
    int Max;
    int now;
};
 
int main() {
    Queue<int> q1(5);
    int a1 = 1, a2 = 2, a3 = 3, a4 = 4, a5 = 5;
    q1.Empty();
    q1.In(a1);
    q1.In(a2);
    q1.In(a3);
    q1.In(a4);
    q1.Empty();
    q1.In(a5);
q1.Empty();
 
    q1.Out(4);
    q1.Out(3);
    q1.Out(2);
    q1.Out(1);
    q1.Out(0);
    return 0;
}
