#include <iostream> #include<new> using namespace std; template<typename T> class Queue { public:     Queue(int max);     Queue(Queue& queue);     ~Queue();     void ResetFront()     {         Front = NULL;     } private:     T* Front;//Point to the first element of the Queue     T* Tail;//Point to the next position of the last element of the Queue     int Max;//Maximum volume of the Queue }; template<typename T> Queue<T>::Queue(int max) {     Front = new(std::nothrow) T[max + 1];     Tail = Front;     Max = max; } template<typename T> Queue<T>::Queue(Queue& queue) {     Front = new(std::nothrow) T[queue.Max + 1];     Tail = Front + (queue.Tail - queue.Front);     Max = queue.Max;     for (int i = 0; i < queue.Tail - queue.Front; i++)     {         *(Front + i) = *(queue.Front + i);     } } template<typename T> Queue<T>::~Queue() {     if (Front == NULL)     {         throw 1;//1 means 'delete a nullptr' error     }     delete[]Front; } int main() {     try     {         Queue<int> p(1);         p.ResetFront();     }     catch (int err)     {         cout << "Error: try to delete a nullptr" << endl;     }     return 0; }
