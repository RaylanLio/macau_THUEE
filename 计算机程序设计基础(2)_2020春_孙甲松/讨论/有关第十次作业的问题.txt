  <p style="margin-left:0cm; margin-right:0cm; text-align:justify"><span style="font-size:10.5pt">应该把：</span></p> 
  <p style="margin-left:0cm; margin-right:0cm; text-align:justify"><span style="font-size:10.5pt">fstream tfile("2.dat");</span></p> 
  <p style="margin-left:0cm; margin-right:0cm; text-align:justify"><span style="font-size:10.5pt">fstream file("22.dat",ios_base::binary);</span></p> 
  <p style="margin-left:0cm; margin-right:0cm; text-align:justify"><span style="font-size:10.5pt">改为：</span></p> 
  <p style="margin-left:0cm; margin-right:0cm; text-align:justify"><span style="font-size:10.5pt">fstream tfile("2.dat",ios_base::out);</span></p> 
  <p style="margin-left:0cm; margin-right:0cm; text-align:justify"><span style="font-size:10.5pt">fstream file("22.dat",ios_base::binary|ios_base::out);</span></p> 
  <p style="margin-left:0cm; margin-right:0cm; text-align:justify"><span style="font-size:10.5pt">或：</span></p> 
  <p style="margin-left:0cm; margin-right:0cm; text-align:justify"><span style="font-size:10.5pt">ofstream tfile("2.dat");</span></p>
