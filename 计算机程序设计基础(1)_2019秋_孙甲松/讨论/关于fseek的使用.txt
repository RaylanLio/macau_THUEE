关于fseek的使用，有程序；
#include <stdio.h>
#include <stdlib.h>
main( )
{ FILE *fp; int k;
double x[5]={1.1, 2.3, 4.5, -3.6, 9.5}, y[6]={0};
if ((fp=fopen("cc.dat","w+b"))==NULL) /* 以二进制写读方式打开文件 */
{ printf("cannot open this file !\n"); exit(0);
}
fwrite(x, sizeof(double), 5, fp);
fseek(fp, 0L, SEEK_SET);
fread(&y[0], sizeof(double), 2, fp);
fseek(fp, -4L*(long)sizeof(double), SEEK_END);
fread(&y[2], sizeof(double), 2, fp);
fseek(fp, - (long)sizeof(double), SEEK_CUR);
fread(&y[4], sizeof(double), 2, fp);
fclose(fp);
for (k=0; k<6; k++) printf("%4.1f ", y[k]);
}
 
能解释一下为什么运行结果为1.1 2.3 2.3 4.5 4.5 -3.6吗?

  <p>课上我不是仔细讲过吗？</p> 
  <p>fwrite(x, sizeof(double), 5, fp);&nbsp;&nbsp; 写出5个double数到文件中，</p> 
  <p>fseek(fp, 0L, SEEK_SET);&nbsp;&nbsp;&nbsp; 把文件指针移到文件开始处，</p> 
  <p>fread(&amp;y[0], sizeof(double), 2, fp);&nbsp;&nbsp; 读入1.1和2.3两个double数到y[0]和y[1]</p> 
  <p>fseek(fp, -4L*(long)sizeof(double), SEEK_END); 把文件指针移到文件尾向前的4个double数开始处，也就是2.3之前，</p> 
  <p>fread(&amp;y[2], sizeof(double), 2, fp);&nbsp;&nbsp; 读入2.3和4.5两个double数到y[2]和y[3]，读完后文件指针处于-3.6开始处，</p> 
  <p>fseek(fp, - (long)sizeof(double), SEEK_CUR);&nbsp;&nbsp; 把文件指针移到当前位置向前的1个double数开始处，也就是4.5之前，</p> 
  <p>fread(&amp;y[4], sizeof(double), 2, fp);&nbsp; 读入4.5和-3.6两个double数到y[4]和y[5]，</p> 
  <p>所以最后输出y数组的结果是： 1.1 2.3 2.3 4.5 4.5 -3.6</p>
