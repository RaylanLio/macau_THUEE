#include<stdio.h> main() {     unsigned long k=0;     scanf("%lu",&k);     while(1)      {         k++;         if (k>=0XFFFFFFFF)              break;      }     printf("end"); } 测速结果：9.02s，8.46s，8.47s，8.49s，8.43s
 
#include<stdio.h> main() {     unsigned long k=0;     scanf("%lu",&k);     for(;;)     {         k++;         if (k>=0XFFFFFFFF)              break;      }     printf("end"); } 测速结果：8.66s，8.58s，8.35s，9.29s，8.47s，8.71s
 
测速方法：键盘输入0，秒表和回车同时按下，出现end立即停止秒表。
测速顺序：1，2，2，2，1，1，1，1，2，2，2
貌似for并没有很明显的优势？
