#include<stdio.h>
#include<stdlib.h>
#pragma warning(disable:4996)
 
struct student
{
    int num;
    char name[15];
    int grade[5];
    int sum;
};//定义结构体
 
int main()
{
    student* p, * Alloc(int n), * sumup(student * p, int n);//函数声明：两个返回指针类型（Alloc和sumup），两个为void类型
    int n = 0;//学生人数n
    void sort(student * p, int n), showinfo(student * p, int n);
 
    printf("Please input the number of student(no more than 20):\n");
    scanf("%d", &n);
    while (n > 20 || n < 0)
    {
        printf("Invalid number!");
        printf("Please input the number of student(no more than 20):\n");
        scanf("%d", &n);
    }//判断n是否满足要求，不满足则重新输入
    p = Alloc(n);//让p指向分配的n个值的地址
    p = sumup(p, n);//计算p中的sum
    printf("\n");
    sort(p, n);
    printf("\n");
    showinfo(p, n);
    return 0;
}
 
student* Alloc(int n)
{
    student* p = (student*)malloc(sizeof(student) * n);//分配一个大小为n的student类型数组
    if (p == NULL)
    {
        printf("Can't get memory!\n");
        exit(1);
    }
    printf("Please input the information of each student in the order of number, name and grades for 5 subjects:\n");
    for (int i = 0; i < n; i++)
    {
        scanf("%d%s", &p[i].num, &p[i].name);//则p[i]为各个数组元素
        for (int j = 0; j < 5; j++)
             scanf("%d", &p[i].grade[j]);
        p[i].sum = 0;
    }
    return p;
}
 
student* sumup(student* p, int n)
{
    for (int j = 0; j < n; j++)
        for (int i = 0; i < 5; i++)
             p[j].sum += p[j].grade[i];
    return p;
}
 
void sort(student* p, int n)
{
    student* w;
    student** q = (student * *)malloc(sizeof(student*) * n);//定义大小为n的指针数组q，让q中的各元素指向p中的各元素
    if (q == NULL)
    {
        printf("Can't get memory!\n");
        exit(0);
    }
    for (int i = 0; i < n; i++)
        q[i] = &p[i];
    for (int i = 0; i < n - 1; i++)
    {
        int k = i;
        for (int j = i + 1; j < n; j++)
             if (q[k]->sum < q[j]->sum)
                 k = j;//选取出当前最小的元素对应的下标k
        if (k != i)
             w = q[i], q[i] = q[k], q[k] = w;//交换指针
    }
    for (int i = 0; i < n; i++)
    {
        printf("%-10d%-10s", q[i]->num, q[i]->name);
        for (int j = 0; j < 5; j++)
             printf("%-4d", q[i]->grade[j]);
        printf("%d\n", q[i]->sum);
    }
    return;
}
 
void showinfo(student* p, int n)
{
    int stu, flag = 1;
    printf("Please input the number of student:\n");
    scanf("%d", &stu);
    for (int i = 0; flag && i < n; i++)
        if (p[i].num == stu)//寻找和输入的学号相同的元素，若找到则置flag为0，输出信息并退出循环
        {
             flag = 0;
             printf("%-10d%-10s", p[i].num, p[i].name);
             for (int j = 0; j < 5; j++)
                 printf("%-4d", p[i].grade[j]);
             printf("%d\n", p[i].sum);
        }
    if (flag)//若退出循环后仍未找到，输出提示信息
        printf("No student with this number!\n");
    return;
}
