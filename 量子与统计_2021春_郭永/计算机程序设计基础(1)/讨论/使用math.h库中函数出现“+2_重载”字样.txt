源代码
#include<stdio.h>
#include<string.h>
#include<math.h>
#pragma warning (disable:4996)
void complex_exp(double *x,double *y)
{
double a=*x,b=*y;
*x=exp(a)*cos(b);
*y=exp(a)*sin(b);
return;
}
 
void complex_ln(double *x,double *y)
{
double a=*x,b=*y;
*x=log(sqrt(a*a+b*b));
*y=atan2(b,a);
return;
}
 
void complex_sin(double *x,double *y)
{
double a=*x,b=*y;
*x=sin(a)*(exp(b)+exp(-b))/2;
*y=cos(a)*(exp(b)-exp(-b))/2;
return;
}
 
void complex_cos(double *x,double *y)
{
double a=*x,b=*y;
*x=cos(a)*(exp(b)+exp(-b))/2;
*y=-sin(a)*(exp(b)-exp(-b))/2;
return;
}
main(int argc,char *argv[])
{
void (*p[])(double *x,double *y)={complex_exp,complex_ln,complex_sin,complex_cos};
double u=2,v=3;
int i;
printf("计算结果是：\n");
for (i=0;i<=3;i++)
{
    p[i](&u,&v);
    printf("%f+j(%f)\n",u,v);
}
return 0;
}
 
 
当鼠标停留在从math.h库中调用的函数上时会出现“+2 重载”等字样
例如，atan2函数显示的就是“double_cdecl atan2(double _Y,double _X) +2 重载”。
对程序运行结果似乎没有影响，但是是因为什么原因呢？
