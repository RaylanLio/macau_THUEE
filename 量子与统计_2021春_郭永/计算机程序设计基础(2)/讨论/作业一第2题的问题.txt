#include <stdio.h> #include <stdlib.h> #include <string.h> #pragma warning(disable:4996)
unsigned char reverse(unsigned char p) {int i; unsigned char r=0; for (i=0;i<8;i++) {if ((p>>i)&1) {r=r|1<<(7-i);} } return r; }
main () {int i=0,k=0; unsigned char a[100]; FILE *fp,*fo; if ((fp=fopen("1.txt","rb"))==NULL) { printf("cannot open this file !\n"); exit(0);} for(i=0;i<100;i++) {if (fscanf(fp,"%c",&a[i])) break; reverse (a[i]);} fclose (fp); if ((fo=fopen("2.txt","wb"))==NULL) { printf("cannot open this file !\n"); exit(0);} for (k=0;k<i;k++) fprintf(fo,"%c",a[k]); fclose (fo); }
 
老师想问一下这个程序为什么运行之后，2.txt显示不了任何东西啊？

孙甲松






 
 
  <p>首先， if (fscanf(fp,"%c",&amp;a[i])) break;</p> 
  <p>错误，fscanf是返回正确读入的数据项数，正确为a[i]读入数据返回值为1，if (fscanf(fp,"%c",&amp;a[i]))为真，执行break;立即终止读入过程，</p> 
  <p>后面的字符都没有读入。</p> 
  <p>第二，reverse (a[i]); 函数调用，并没有改变a[i]的值，逆转后的值是通过reverse函数值返回的，应改为：&nbsp;a[i] =&nbsp;reverse (a[i]);</p> 
  <p>第三，既然是用fscanf和fprintf函数读写文件，打开文件时就不应该以二进制方式打开，按普通文本方式打开就行了。</p>
