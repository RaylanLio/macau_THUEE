 /*老师您好，想请问一下为什么不能用cout<<*itr.size()<<endl;，对iterator解引用返回的是什么呢？*/
#include<iostream> #include<cstdlib> #include<string> #include<vector> #include<iterator> using namespace std;
int main(void) {   
vector<string> vi;     string ss;     while(cin>>ss)     {         vi.push_back(ss);//用空格隔开，ctrl-z结束输入     }     for(vector<string>::iterator itr=vi.begin();itr!=vi.end();++itr)     {                  string temp;         temp=*itr;         cout<<temp.size()<<endl;         /*为什么不能用cout<<*itr.size()<<endl;*/     }     return 0;     }
