代码如下：
#include <iostream>
#include <math.h>
#include <string.h>
using namespace std;
class String{
private:
    char *str;
public:
    String(char *c="0");
    String(const String &p);
    ~String(){
       // cout<<"析构函数被调用！"<<endl;
    };
    void Set(const char *p);
    void Print();
    char* Getstr(){
        return str;
    }
    String& operator=(String str2){
        if (&str2!=this) {
            delete []this->str;
            this->str=new char[strlen(str2.str)+1];
            strcpy(this->str, str2.str);
        }
        return *this;
    }
    friend String operator+(String &str1,String &str2);
};
String::String(char *c){
    this->str=new char[strlen(c)+1];
    strcpy(str, c);

    this->str=new char[strlen(p.str)+1];
    strcpy(this->str, p.str);

    delete []str;

}

  <p>是因为你的程序中类的友元声明：</p> 
  <p>&nbsp; &nbsp;&nbsp;&nbsp; &nbsp;&nbsp;<strong>friend</strong>&nbsp;String&nbsp;<strong>operator</strong>+(String &amp;str1,String &amp;str2);</p> 
  <p>与你的与友元函数的定义：</p> 
  <p>String&nbsp;<strong>operator</strong>+( String str1,String str2){</p> 
  <p>&nbsp; &nbsp;&nbsp;<strong>char</strong>&nbsp;*q=<strong>new</strong>&nbsp;<strong>char</strong>[strlen(str1.Getstr())+strlen(str2.Getstr())+1];</p> 
  <p>&nbsp; &nbsp; strcpy(q, str1.Getstr());</p> 
  <p>&nbsp; &nbsp; strcat(q, str2.Getstr());</p> 
  <p>&nbsp; &nbsp; String S(q);</p> 
  <p>&nbsp; &nbsp;&nbsp;<strong>return</strong>&nbsp;S;</p> 
  <p>}</p> 
  <p>函数头部不一致导致了编译错误，你把函数头部改为：</p> 
  <p>String&nbsp;<strong>operator</strong>+( String <span style="color:#e74c3c">&amp;</span>str1,String <span style="color:#e74c3c">&amp;</span>str2){</p> 
  <p>就没有问题了。</p> 
  <p>你的函数中在 return S; 之前，应该加上&nbsp; delete []q;&nbsp;</p> 
  <p>否则会有内存泄漏。</p>
