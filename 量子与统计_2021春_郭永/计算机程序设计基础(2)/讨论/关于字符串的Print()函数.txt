#include<iostream>
#include<cstring>
using namespace std;
class String
{
public:
    String(const char * s="0");
    ~String();
    void Set(const char *s);
    String(const String & s);
    void Print();
    friend bool operator == (String &s1,String &s2);
    friend bool operator > (String &s1,String &s2);
    friend bool operator < (String &s1,String &s2);
    friend bool operator != (String &s1,String &s2);
    friend bool operator >= (String &s1,String &s2);
    friend bool operator <= (String &s1,String &s2);
private:
    char * p;
};
String :: String(const char * s)
{
    p=new char[strlen(s)+1];
    strcpy(p,s);
    cout<<"构造函数被调用！"<<endl;
    Print();
}
String :: ~String()
{
    delete []p;
    cout<<"析构函数被调用!"<<endl;
    Print();
}
void String :: Set(const char *s)
{
    delete []p;
    p=new char [strlen(s)+1];
    strcpy(p,s);
    cout<<"设置函数被调用"<<endl;
    Print();
}
void String :: Print()
{
    cout<<"String: "<<p<<endl;
}
String :: String(const String & s)
{
    this->p=new char[strlen(s.p)+1];
    strcpy(this->p,s.p);
    cout<<"复制函数被调用！"<<endl;
    Print();
}
bool operator == (String &s1,String &s2)
{
    if(strcmp(s1.p,s2.p)==0)
         return true;
    else 
         return false;
}
bool operator > (String &s1,String &s2)
{
    if(strcmp(s1.p,s2.p)>0)
         return true;

}
bool operator < (String &s1,String &s2)
{
    if(strcmp(s1.p,s2.p)<0)
         return true;
    else 
         return false;
}
bool operator != (String &s1,String &s2)
{
    if(strcmp(s1.p,s2.p)!=0)
         return true;
    else 
         return false;
}
bool operator >= (String &s1,String &s2)
{
    if(strcmp(s1.p,s2.p)>=0)
         return true;
    else 
         return false;
}
bool operator <= (String &s1,String &s2)
{
    if(strcmp(s1.p,s2.p)<=0)
         return true;
    else 
         return false;
}

  <p style="margin-left:0cm; margin-right:0cm; text-align:justify"><span style="font-size:10.5pt">其实是最后执行析构函数打印字符串时出现了乱码，因为你的析构函数：</span></p> 
  <p style="margin-left:0cm; margin-right:0cm; text-align:justify"><span style="font-size:10.5pt">String :: ~String()</span></p> 
  <p style="margin-left:0cm; margin-right:0cm; text-align:justify"><span style="font-size:10.5pt">{</span></p> 
  <p style="margin-left:0cm; margin-right:0cm; text-align:justify"><span style="font-size:10.5pt">&nbsp;&nbsp;&nbsp; delete []p;</span></p> 
  <p style="margin-left:0cm; margin-right:0cm; text-align:justify"><span style="font-size:10.5pt">&nbsp;&nbsp;&nbsp; cout&lt;&lt;"析构函数被调用!"&lt;&lt;endl;</span></p> 
  <p style="margin-left:0cm; margin-right:0cm; text-align:justify"><span style="font-size:10.5pt">&nbsp;&nbsp;&nbsp; Print();</span></p> 
  <p style="margin-left:0cm; margin-right:0cm; text-align:justify"><span style="font-size:10.5pt">}</span></p> 
  <p style="margin-left:0cm; margin-right:0cm; text-align:justify"><span style="font-size:10.5pt">中先用delete释放了p所指的字符串动态内存块，然后用Print();打印字符串，所以打印出来的是乱码，应该先打印后释放动态内存：</span></p> 
  <p style="margin-left:0cm; margin-right:0cm; text-align:justify"><span style="font-size:10.5pt">String :: ~String()</span></p> 
  <p style="margin-left:0cm; margin-right:0cm; text-align:justify"><span style="font-size:10.5pt">{</span></p> 
  <p style="margin-left:0cm; margin-right:0cm; text-align:justify"><span style="font-size:10.5pt">&nbsp;&nbsp;&nbsp; cout&lt;&lt;"析构函数被调用!"&lt;&lt;endl;</span></p> 
  <p style="margin-left:0cm; margin-right:0cm; text-align:justify"><span style="font-size:10.5pt">&nbsp;&nbsp;&nbsp; Print();</span></p> 
  <p style="margin-left:0cm; margin-right:0cm; text-align:justify"><span style="font-size:10.5pt">&nbsp;&nbsp;&nbsp; delete []p;</span></p> 
  <p style="margin-left:0cm; margin-right:0cm; text-align:justify"><span style="font-size:10.5pt">}</span></p>
